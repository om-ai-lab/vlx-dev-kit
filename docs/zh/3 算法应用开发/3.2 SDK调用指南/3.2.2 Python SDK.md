## 对接方式

### 方式一：使用 .tar.gz 包

使用 .tar.gz 包时，业务项目通过 pip install 安装 SDK，然后从 omai_client 包中导入 VlxApiClient。SDK 会通过 VlxApiClient.from_env() 读取环境变量配置并创建客户端实例。

请至[钉钉文档](https://alidocs.dingtalk.com/i/nodes/N7dx2rn0Jb6dRYeGhZw3vmwoJMGjLRb3)查看附件《omai-1.0.3.tar.gz》。

### 方式二：使用pypi中已推送的omai包
使用已经推送到官方pypi库中的包安装SDK，通过pip install omai安装，可以指定对应版本

**先安装：**

```bash
pip install ./omai-1.0.3.tar.gz
```
或
```bash
pip install omai
```

**业务代码中引入：**

```python
from omai_client import VlxApiClient

# 从环境变量读取配置，创建客户端
vlx_api_client = VlxApiClient.from_env()
```

下面的所有方法都支持自己传输 `Map<String, String>` headers 。如果key中包含 Authorization 就不会再一句配置鉴权

```python
headers = {
    "Authorization": "Bearer your_custom_token",
};
```

**使用：**

## 调用 VLX-Seek

```python
from omai_client import (
    VlxApiClient,
    VlxSeekRequest,
    VlxSeekMessage,
    VlxSeekContent,
    VlxSeekImageUrl,
)


# 请求 vlx-seek
image_url = VlxSeekImageUrl(
    url="https://example.com/image.jpg"
)

image = VlxSeekContent(
    type="image_url",
    image_url=image_url,
)

text = VlxSeekContent(
    type="text",
    text="请描述图片内容",
)

message = VlxSeekMessage(
    role="user",
    content=[image, text],
)

# bbox_list: [[28, 54, 481, 387]]
bbox_list = [
    [28, 54, 481, 387]
]

request = VlxSeekRequest(
    model="vlx-seek-1.5-3b",
    messages=[message],
    bbox_list=bbox_list,
    max_tokens=4096,
    temperature=0.0,
    top_p=1.0,
)

response = vlx_api_client.vlx_seek(request)
```

## 调用 VLX-FLow
### 方式一：
Python 对应 方法一 是：消费 SSE 外层事件，也就是拿到 event.event、event.data、event.retry、event.id，然后业务方自己解析 event.data。

```python
//请求发起vlx-flow 长流式

import json

from omai_client import (
    VlxApiClient,
    VlxFlowSource,
    VlxFlowStreamRequest,
    VlxFlowStopRequest,
    VlxStreamEventMessage,
)


# 请求参数组装
source = VlxFlowSource(
    path="https://om-agent-test.linker.cc/minio/goal/cover/vlxflow-video01.mp4",
    type=2,
)

request = VlxFlowStreamRequest(
    model="VLX-Flow",
    prompt="实时描绘空中全景画面，字数 20 字以内",
    system_prompt="",
    interval_seconds=1.0,
    frame_preempt="queue",
    temperature=0.7,
    max_tokens=40,
    source=source,
)

# 自定义请求头，可选；不需要就传 None 或直接不传
headers = None


class VlxFlowHandler:
    def __init__(self):
        self.task_id = None

    def on_event(self, event):
        # SSE 外层事件名，例如 message、done、error、ping 等。
        print("sse event=", event.event)

        # SSE data 字段，服务端业务消息 JSON 字符串。
        data = event.data
        print("sse data=", data)

        if not data:
            return

        # 如果业务侧需要处理 VLX 业务内容，需要自行解析 data。
        message = VlxStreamEventMessage.from_dict(json.loads(data))

        # 保存 taskId，后续如果要停止服务端任务，需要调用 vlx_flow_stop。
        if message.taskId is not None and self.task_id is None:
            self.task_id = int(message.taskId)

        print("business event=", message.event)
        print("content=", message.content)

    def on_error(self, throwable):
        # HTTP 异常、读流异常、解析异常会进入这里。
        print("vlx stream error:", throwable)

    def on_closed(self):
        # 长连接关闭时回调。正常结束、异常结束、本地 cancel 都会触发。
        print("vlx stream closed")


handler = VlxFlowHandler()

# 调用 vlx-flow 长流式，方法一：消费 SSE 外层事件
call = vlx_api_client.vlx_flow_stream(request, handler, headers=headers)

# 只取消本地 SSE 长连接，不等于停止服务端 VLX 任务。
# 注意：真实业务中通常是在需要停止接收时再调用，不要刚发起就立刻 cancel。
call.cancel()

# 如需停止服务端任务，需要拿 taskId 调停止接口。
if handler.task_id is not None:
    vlx_api_client.vlx_flow_stop(
        VlxFlowStopRequest(
            taskId=handler.task_id,
        )
    )
call = vlx_api_client.vlx_flow_stream(request, handler)
```

### 方式二：
Python 对应 方法二（推荐） 是：直接消费 VLX 业务事件。SDK 内部会把 SSE 的 data JSON 解析成 VlxStreamEventMessage，业务侧不需要自己解析 SSE 外层结构。

```python
//请求发起vlx-flow 长流式
from omai_client import (
    VlxApiClient,
    VlxFlowSource,
    VlxFlowStreamRequest,
    VlxFlowStopRequest,
)


# 请求参数组装
source = VlxFlowSource(
    path="https://om-agent-test.linker.cc/minio/goal/cover/vlxflow-video01.mp4",
    type=2,
)

request = VlxFlowStreamRequest(
    model="VLX-Flow",
    prompt="实时描绘空中全景画面，字数 20 字以内",
    system_prompt="",
    interval_seconds=1.0,
    frame_preempt="queue",
    temperature=0.7,
    max_tokens=40,
    source=source,
)

# 自定义请求头，可选；不需要就传 None 或调用时不传 headers
headers = None


class VlxFlowMessageHandler:
    def __init__(self):
        self.task_id = None

    def on_event(self, message):
        # SDK 内部已经把 SSE data 解析成 VlxStreamEventMessage。
        # 业务侧直接处理 VLX 业务事件即可。

        if message.taskId is not None and self.task_id is None:
            self.task_id = int(message.taskId)

        # 业务事件类型，例如 taskCreated、delta、lineBreak、done、error、stopped。
        event = message.event

        print("taskId=", message.taskId)
        print("event=", event)
        print("frameNo=", message.frameNo)
        print("content=", message.content)

        if event == "delta":
            # 增量内容，可以累积到字符串、写 DB、转发给前端 SSE 等。
            print("增量输出：", message.content)

        if event == "done":
            # 服务端任务结束。
            print("VLX 任务结束")

        if event == "error":
            # 服务端业务异常。
            print("VLX 任务异常：", message.content)

    def on_error(self, throwable):
        # 连接异常或 data 解析异常。
        print("vlx message stream error:", throwable)

    def on_closed(self):
        # 长连接已关闭，可释放业务侧资源。
        print("vlx message stream closed")


handler = VlxFlowMessageHandler()

# 调用 vlx-flow 长流式，方法二：直接消费 VLX 业务事件（推荐）
call = vlx_api_client.vlx_flow_stream_message(request, handler, headers=headers)

# 只关闭本地长连接，不等于停止服务端任务。
# 注意：真实业务中通常是在需要停止接收时再调用，不要刚发起就立刻 cancel。
call.cancel()

# 停止服务端任务。
if handler.task_id is not None:
    vlx_api_client.vlx_flow_stop(
        VlxFlowStopRequest(
            taskId=handler.task_id,
        )
    )
call = vlx_api_client.vlx_flow_stream_message(request, handler)
```

### 方式三：
Python 对应 方法三 是：直接拿原始 HTTP 流式响应，自己按 SSE 行解析。这个方式也不推荐，业务侧要自己处理阻塞、关闭连接、解析 SSE、异常。

方法三是原始流读取方式，不推荐。Python SDK 不会为该方法启动后台线程，也不会自动回调业务事件；如果只想处理 VLX 业务消息，推荐使用 vlx_flow_stream_message。

```python
//请求发起vlx-flow 长流式
import json

from omai_client import (
    VlxApiClient,
    VlxFlowSource,
    VlxFlowStreamRequest,
    VlxSseEventParser,
    VlxStreamEventMessage,
)


# 请求参数组装
source = VlxFlowSource(
    path="https://om-agent-test.linker.cc/minio/goal/cover/vlxflow-video01.mp4",
    type=2,
)

request = VlxFlowStreamRequest(
    model="VLX-Flow",
    prompt="实时描绘空中全景画面，字数 20 字以内",
    system_prompt="",
    interval_seconds=1.0,
    frame_preempt="queue",
    temperature=0.7,
    max_tokens=40,
    source=source,
)

# 自定义请求头，可选；不需要就传 None 或调用时不传 headers
headers = None

# Raw 方法不会帮你开后台线程，也不会帮你解析 SSE。
# 这个调用会发起原始 HTTP 流式请求，后续读取 response.iter_lines() 是阻塞的。
response = None

try:
    response = vlx_api_client.vlx_flow_stream_raw(request, headers=headers)

    if response.status_code < 200 or response.status_code >= 300:
        raise RuntimeError(f"vlx stream http error, code={response.status_code}")

    parser = VlxSseEventParser()

    for line in response.iter_lines(decode_unicode=True):
        # parser 会按 SSE 协议聚合多行 data，遇到空行才返回完整 VlxFlowEvent。
        event = parser.accept_line(line)
        if event is None:
            continue

        # 外层 SSE 事件。
        print("sse event=", event.event)

        # 内层 VLX 业务消息。
        data = event.data
        if not data:
            continue

        message = VlxStreamEventMessage.from_dict(json.loads(data))
        print("business event=", message.event)
        print("content=", message.content)

    # 如果连接关闭前还有未以空行结尾的事件，可以 flush 一次。
    last_event = parser.complete()
    if last_event is not None and last_event.data:
        message = VlxStreamEventMessage.from_dict(json.loads(last_event.data))
        print("last business event=", message.event)

except Exception as e:
    # 主动关闭 response 也可能导致读取异常，这类通常不用当作业务异常。
    raise RuntimeError(e)

finally:
    # 取消/关闭原始 HTTP 长连接。
    if response is not None:
        response.close()
response = vlx_api_client.vlx_flow_stream_raw(request)
from omai_client import (
    VlxApiClient,
    VlxFlowStopRequest,
    VlxFlowUpdateConfigRequest,
)


//停止vlx-flow
request = VlxFlowStopRequest(
    taskId=888,
)

response = vlx_api_client.vlx_flow_stop(request)



//更新vlx-flow
request = VlxFlowUpdateConfigRequest(
    taskId=888,
    prompt="识别画面中是否有人进入危险区域",
    frame_preempt="preempt",
)

response = vlx_api_client.vlx_flow_update_config(request)

//如果要取返回结果：
success = response.data
```
