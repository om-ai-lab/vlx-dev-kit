```xml
<dependency>

    <artifactId>omai-sdk-java</artifactId>
```

## VLX Client (Java JDK 8+)

**Configuration:**

```yaml
# VLX request
vlx:
  api:
    # Enable or disable
    enabled: true
    # Gateway address; append os/ to the endpoint obtained below
    endpoint: http://linker-gateway-service:31003/os/
    # AaaS tenant access token
    api-key: aaas_ondoYHbrL0IGkA7bXMEPsFKCtsQuPhFXLuB0XBXKukwtTNSoZ447EiEwkgDvchJB
    # Connection timeout (ms)
    connect-timeout-ms: 60000
    # Read timeout (ms)
    read-timeout-ms: 60000
    # Write timeout (ms)
    write-timeout-ms: 60000
```

**Injection**

```java
@Resource
private VlxApiClient vlxApiClient;
```

All methods accept custom `Map<String, String>` headers. If the key contains `Authorization`, the configured credential is not added.

## Call VLX-Flow


```java
// Start a long-running vlx-flow request
// Build request parameters
VlxFlowStreamRequest.Source source = VlxFlowStreamRequest.Source.builder()
        .path("https://om-agent-test.linker.cc/minio/goal/cover/vlxflow-video01.mp4")
        .type(2)
        .build();

VlxFlowStreamRequest request = VlxFlowStreamRequest.builder()
        .model("VLX-Flow")
        .prompt("Describe the aerial panoramic view in real time, within 20 words")
        .system_prompt("")
        .interval_seconds(1D)
        .frame_preempt("queue")
        .temperature(0.7D)
        .max_tokens(40)
        .source(source)
        .build();

// Call vlx-flow (method 1)

AtomicReference<Long> taskIdRef = new AtomicReference<>();

// Use when you need outer SSE fields: id / event / data / retry.
// Note: event.getData() contains the VLX business-message JSON.
VlxFlowStreamCall call = vlxApiClient.vlxFlowStream(request, new VlxFlowEventHandler() {
    @Override
    public void onEvent(VlxFlowEvent event) {
        // Outer SSE event name, such as message, done, error, or ping.
        System.out.println("sse event=" + event.getEvent());

        // SSE data field containing the server business-message JSON string.
        String data = event.getData();
        System.out.println("sse data=" + data);

        if (data == null || data.isEmpty()) {
            return;
        }

        // Parse data here if application code needs the VLX business payload.
        VlxStreamEventMessage message = JSON.parseObject(data, VlxStreamEventMessage.class);

        // Save taskId; call vlxFlowStop later to stop the server task.
        if (message.getTaskId() != null) {
            taskIdRef.compareAndSet(null, Long.valueOf(message.getTaskId()));
        }

        System.out.println("business event=" + message.getEvent());
        System.out.println("content=" + message.getContent());
    }

    @Override
    public void onError(Throwable throwable) {
        // HTTP, stream-read, and parsing errors arrive here.
        throwable.printStackTrace();
    }

    @Override
    public void onClosed() {
        // Called when the long connection closes (normal, error, or local cancel).
        System.out.println("vlx stream closed");
    }
}, headers);

// Cancels only the local SSE connection; it does not stop the server task.
call.cancel();

// To stop the server task, call the stop API with taskId.
Long taskId = taskIdRef.get();
if (taskId != null) {
    vlxApiClient.vlxFlowStop(VlxFlowStopRequest.builder()
            .taskId(taskId)
            .build());
}
```

```java
AtomicReference<Long> taskIdRef = new AtomicReference<>();

// The SDK parses SSE data into VlxStreamEventMessage.
// Application code can handle VLX business events directly.
VlxFlowStreamCall call = vlxApiClient.vlxFlowStreamMessage(request, new VlxFlowEventMessageHandler() {
    @Override
    public void onEvent(VlxStreamEventMessage message) {
        if (message.getTaskId() != null) {
            taskIdRef.compareAndSet(null, Long.valueOf(message.getTaskId()));
        }

        // Business event types include taskCreated, delta, lineBreak, done, error, and stopped.
        String event = message.getEvent();

        System.out.println("taskId=" + message.getTaskId());
        System.out.println("event=" + event);
        System.out.println("frameNo=" + message.getFrameNo());
        System.out.println("content=" + message.getContent());

        if ("delta".equals(event)) {
            // Accumulate incremental content, write it to a DB, or forward it to frontend SSE.
            System.out.println("Incremental output: " + message.getContent());
        }

        if ("done".equals(event)) {
            // Server task completed.
            System.out.println("VLX task completed");
        }

        if ("error".equals(event)) {
            // Server-side business error.
            System.out.println("VLX task error: " + message.getContent());
        }
    }

    @Override
    public void onError(Throwable throwable) {
        // Connection or data-parsing error.
        throwable.printStackTrace();
    }

    @Override
    public void onClosed() {
        // Long connection closed; release application resources.
        System.out.println("vlx message stream closed");
    }
}, headers);

// Close only the local long connection.
call.cancel();

// Stop the server task.
Long taskId = taskIdRef.get();
if (taskId != null) {
    vlxApiClient.vlxFlowStop(VlxFlowStopRequest.builder()
            .taskId(taskId)
            .build());
}
```

```java
// Start a long-running vlx-flow request
// Build request parameters
VlxFlowStreamRequest.Source source = VlxFlowStreamRequest.Source.builder()
        .path("https://om-agent-test.linker.cc/minio/goal/cover/vlxflow-video01.mp4")
        .type(2)
        .build();

VlxFlowStreamRequest request = VlxFlowStreamRequest.builder()
        .model("VLX-Flow")
        .prompt("Describe the aerial panoramic view in real time, within 20 words")
        .system_prompt("")
        .interval_seconds(1D)
        .frame_preempt("queue")
        .temperature(0.7D)
        .max_tokens(40)
        .source(source)
        .build();


// Call vlx-flow (method 3, not recommended; no taskId, cancellation only)

// The raw method does not start a background thread or parse SSE.
// call.execute() is blocking; run it in a thread pool.
Call<ResponseBody> call = vlxApiClient.vlxFlowStreamRaw(request, headers);

try {
    Response<ResponseBody> response = call.execute();

    if (!response.isSuccessful()) {
        throw new RuntimeException("vlx stream http error, code=" + response.code());
    }

    ResponseBody body = response.body();
    if (body == null) {
        throw new RuntimeException("vlx stream body is null");
    }

    VlxSseEventParser parser = new VlxSseEventParser();

    try (BufferedReader reader = new BufferedReader(
            new InputStreamReader(body.byteStream(), StandardCharsets.UTF_8))) {

        String line;
        while ((line = reader.readLine()) != null) {
            // The parser aggregates data lines per SSE and returns a complete event at a blank line.
            VlxFlowEvent event = parser.acceptLine(line);
            if (event == null) {
                continue;
            }

            // Outer SSE event.
            System.out.println("sse event=" + event.getEvent());

            // Inner VLX business message.
            String data = event.getData();
            if (data == null || data.isEmpty()) {
                continue;
            }

            VlxStreamEventMessage message = JSON.parseObject(data, VlxStreamEventMessage.class);
            System.out.println("business event=" + message.getEvent());
            System.out.println("content=" + message.getContent());
        }
    }

    // Flush once if an event without a trailing blank line remains before close.
    VlxFlowEvent lastEvent = parser.complete();
    if (lastEvent != null && lastEvent.getData() != null) {
        VlxStreamEventMessage message =
                JSON.parseObject(lastEvent.getData(), VlxStreamEventMessage.class);
        System.out.println("last business event=" + message.getEvent());
    }
} catch (IOException e) {
    // Explicit cancellation may raise IOException; this is usually not a business error.
    if (!call.isCanceled()) {
        throw new RuntimeException(e);
    }
}

// Cancel the raw HTTP connection.
call.cancel();
```
